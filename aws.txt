# ==========================================
# PART 3: Connect to AWS EC2 Instance (Windows CMD as Administrator)
# ==========================================

# 1. Navigate to the folder containing your downloaded .pem key file [00:07:03]
cd <path-to-folder-with-pem-file>

# 2. SSH into your Ubuntu EC2 instance (copy exact command from AWS EC2 Connect -> SSH client) [00:07:19]
ssh -i "MyExampleKeyPair.pem" ubuntu@<your-ec2-public-dns>.compute-1.amazonaws.com


# ==========================================
# PART 4: Setup & Deploy on Ubuntu EC2 Terminal
# ==========================================

# 1. Update Ubuntu packages and install Docker, Git, and Nano [00:07:38]
sudo apt update
sudo apt-get install docker.io
sudo apt install git
sudo apt install nano

# 2. Clone your GitHub repository onto the EC2 instance [00:10:59]
git clone https://github.com/<your-username>/<your-repo-name>.git
ls
cd <your-repo-name>
ls

# 3. Create the Dockerfile using Nano [00:11:33]
nano Dockerfile

# Paste the following two lines inside Nano, then save (Ctrl+O, Enter) and exit (Ctrl+X) [00:11:51]:
FROM nginx:alpine
COPY . /usr/share/nginx/html

# 4. Build the Docker image [00:12:22]
sudo docker build -t mywebapp .

# 5. Run the Docker container in detached mode on port 80 [00:12:48]
sudo docker run -d -p 80:80 mywebapp

# Browser URL: http://<your-ec2-public-ipv4> [00:13:34]


# ==========================================
# PART 5: Stop Container & Cleanup
# ==========================================

# 1. Check running containers and copy the Container ID [00:14:00]
sudo docker ps

# 2. Stop the running container [00:14:19]
sudo docker stop <container-id>

# (Remember to also Terminate the EC2 instance and click End Lab in AWS Console) [00:14:44]
